cmake_minimum_required(VERSION 3.20)
project(IntegrityGuardConsumer C)
set(INTEGRITY_GUARD_LIBRARY "" CACHE FILEPATH "NativeAOT import library")
if(NOT EXISTS "${INTEGRITY_GUARD_LIBRARY}")
  message(FATAL_ERROR "Set INTEGRITY_GUARD_LIBRARY to IntegrityGuard.Native.lib")
endif()
add_executable(integrity_guard_sample main.c)
target_include_directories(integrity_guard_sample PRIVATE ../../native/include)
target_link_libraries(integrity_guard_sample PRIVATE "${INTEGRITY_GUARD_LIBRARY}")
target_compile_features(integrity_guard_sample PRIVATE c_std_11)
if(MSVC)
  target_compile_definitions(integrity_guard_sample PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
