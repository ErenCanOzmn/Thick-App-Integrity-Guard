cmake_minimum_required(VERSION 3.20)
project(IntegrityGuardLua C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This release targets Windows x64 only.")
endif()
find_package(Lua 5.4 EXACT REQUIRED)
set(INTEGRITY_GUARD_LIBRARY "" CACHE FILEPATH "NativeAOT-generated IntegrityGuard.Native.lib")
if(NOT EXISTS "${INTEGRITY_GUARD_LIBRARY}")
  message(FATAL_ERROR "Set INTEGRITY_GUARD_LIBRARY to the native import library.")
endif()
add_library(integrity_guard MODULE integrity_guard_lua.c)
target_compile_features(integrity_guard PRIVATE c_std_11)
target_include_directories(integrity_guard PRIVATE ${LUA_INCLUDE_DIR} ../../native/include)
target_link_libraries(integrity_guard PRIVATE ${LUA_LIBRARIES} "${INTEGRITY_GUARD_LIBRARY}")
set_target_properties(integrity_guard PROPERTIES PREFIX "")
